Google Sheets Integration

This allows your logs to instantly post to a live Google Sheet. Whenever you click Save Incident, your dog's episode is securely piped into your sheet.

Apps Script Code Template:

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Create header row if empty
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Incident ID", "Date", "Time", "Period", 
        "Location", "Posture", "Severity", 
        "Symptoms", "Diet", "Time Since Food", 
        "Diet Notes", "Medication", "Duration", 
        "Vet Notes", "Last Updated"
      ]);
      sheet.getRange(1, 1, 1, 15).setFontWeight("bold");
    }
    
    var rowValues = [
      data.id || "",
      data.date || "",
      data.time || "",
      data.timePeriod || "",
      data.location || "",
      data.posture || "",
      data.severity || "",
      (data.symptoms || []).join(", "),
      data.diet || "",
      data.timeSinceFood || "",
      data.dietNotes || "",
      data.medication || "",
      data.duration || "",
      data.notes || "",
      new Date().toISOString()
    ];

    // Search for existing row by Incident ID (Column 1 / A)
    var targetRow = -1;
    var lastRow = sheet.getLastRow();
    if (lastRow > 1 && data.id) {
      var idColumnValues = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
      for (var i = 0; i < idColumnValues.length; i++) {
        if (String(idColumnValues[i][0]).trim() === String(data.id).trim()) {
          targetRow = i + 2; // +2 for 1-based index and skipping header row
          break;
        }
      }
    }

    if (targetRow > 0) {
      // Overwrite existing row
      sheet.getRange(targetRow, 1, 1, rowValues.length).setValues([rowValues]);
    } else {
      // Append as new incident
      sheet.appendRow(rowValues);
    }
    
    return ContentService
      .createTextOutput(JSON.stringify({ status: "success", action: targetRow > 0 ? "updated" : "created" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
